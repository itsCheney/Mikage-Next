cmake_minimum_required(VERSION 3.20)
project(TJSNativeFactoryTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../../Engine/KRKRRuntime/Source/cpp" REALPATH)
file(GLOB TJS_SOURCES CONFIGURE_DEPENDS "${CORE}/tjs2/*.cpp")
add_executable(tjs-native-factory-tests NativeFactoryTests.cpp ${TJS_SOURCES})
target_include_directories(tjs-native-factory-tests PRIVATE "${CORE}/tjs2" "${CORE}/environ")
target_compile_definitions(tjs-native-factory-tests PRIVATE TJS_NO_REGEXP=1 _ONLYCONSOLE=1)
if(WIN32)
    target_compile_definitions(tjs-native-factory-tests PRIVATE _KRKRSDL3_WINDOWS=1)
elseif(APPLE)
    target_compile_definitions(tjs-native-factory-tests PRIVATE _KRKRSDL3_MACOS=1)
else()
    target_compile_definitions(tjs-native-factory-tests PRIVATE _KRKRSDL3_LINUX=1)
endif()
enable_testing()
add_test(NAME tjs-native-factory-tests COMMAND tjs-native-factory-tests)
