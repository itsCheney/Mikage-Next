cmake_minimum_required(VERSION 3.20)
project(AlphaMovieTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../../Engine/KRKRRuntime/Source/cpp" REALPATH)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(ZLIB REQUIRED)
set(SOURCE "${CORE}/plugins/AlphaMovie.cpp")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/ProductionAlphaMovie.inc")
set(CODEC "${CMAKE_CURRENT_BINARY_DIR}/ProductionAMVCodec.inc")
add_custom_command(OUTPUT "${GENERATED}" "${CODEC}" COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/extract-production.py" --source "${SOURCE}" --output "${GENERATED}" --codec-output "${CODEC}"
    DEPENDS "${SOURCE}" extract-production.py VERBATIM)
add_executable(alpha-movie-tests AlphaMovieTests.cpp "${GENERATED}")
target_include_directories(alpha-movie-tests PRIVATE "${CORE}/plugins" "${CORE}/core/render" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(alpha-movie-tests PRIVATE ZLIB::ZLIB)
enable_testing()
add_test(NAME alpha-movie-tests COMMAND alpha-movie-tests)
add_executable(alpha-movie-codec-tests AlphaMovieCodecTests.cpp "${GENERATED}" "${CODEC}")
target_include_directories(alpha-movie-codec-tests PRIVATE "${CORE}/plugins" "${CORE}/core/render" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(alpha-movie-codec-tests PRIVATE ZLIB::ZLIB)
add_test(NAME alpha-movie-codec-tests COMMAND alpha-movie-codec-tests)
